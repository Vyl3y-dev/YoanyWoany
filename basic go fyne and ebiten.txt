func NewCard(title, subtitle string) fyne.CanvasObject {
    t := canvas.NewText(title, color.White)
    t.TextSize = 20
    s := canvas.NewText(subtitle, color.Gray{Y:180})

    box := container.NewVBox(t, s)
    padded := container.NewPadded(box)

    bg := canvas.NewRectangle(color.RGBA{45,45,55,255})
    bg.SetMinSize(fyne.NewSize(250,120))

    return container.NewMax(bg, padded)
}

9. Sprite Animation
-------------------
type Anim struct{
    Frames []*ebiten.Image
    Index int
    Tick int
}

func (a *Anim) Update(){
    a.Tick++
    if a.Tick%6==0 {
        a.Index = (a.Index+1)%len(a.Frames)
    }
}

func (a *Anim) Draw(screen *ebiten.Image, x,y float64){
    op := &ebiten.DrawImageOptions{}
    op.GeoM.Translate(x,y)
    screen.DrawImage(a.Frames[a.Index], op)
}

10. Collision

func Collide(x1,y1,w1,h1,x2,y2,w2,h2 float64) bool {
    return x1 < x2+w2 &&
           x1+w1 > x2 &&
           y1 < y2+h2 &&
           y1+h1 > y2
}

1. time — Working With Dates, Durations, Timers
============================================================

Current time:
--------------
now := time.Now()

Format (Go uses reference date 2006-01-02):
-------------------------------------------
fmt.Println(now.Format("2006-01-02 15:04"))

Sleeping:
---------
time.Sleep(2 * time.Second)

Timer:
------
timer := time.NewTimer(3 * time.Second)
<-timer.C   // waits 3 seconds

Ticker (repeats):
-----------------
ticker := time.NewTicker(1 * time.Second)
for range ticker.C {
    fmt.Println("tick")
}

Add/Subtract time:
------------------
future := now.Add(48 * time.Hour)
past := now.Add(-1 * time.Hour)

Compare:
--------
if a.Before(b) {}
if a.After(b)  {}
if a.Equal(b)  {}

Duration:
---------
d := time.Since(startTime)
