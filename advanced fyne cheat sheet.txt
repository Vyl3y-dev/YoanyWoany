1. CUSTOM WIDGETS
-----------------------------------
Define your own widget type:

type MyWidget struct {
    widget.BaseWidget
    text string
}

func NewMyWidget(t string) *MyWidget {
    w := &MyWidget{text: t}
    w.ExtendBaseWidget(w)
    return w
}

func (w *MyWidget) CreateRenderer() fyne.WidgetRenderer {
    lbl := canvas.NewText(w.text, color.White)

    return &myWidgetRenderer{
        widget: w,
        objects: []fyne.CanvasObject{lbl},
        label: lbl,
    }
}

type myWidgetRenderer struct {
    widget  *MyWidget
    objects []fyne.CanvasObject
    label   *canvas.Text
}

func (r *myWidgetRenderer) Layout(size fyne.Size) {}
func (r *myWidgetRenderer) MinSize() fyne.Size   { return r.label.MinSize() }
func (r *myWidgetRenderer) Refresh()             { canvas.Refresh(r.label) }
func (r *myWidgetRenderer) Destroy()             {}
func (r *myWidgetRenderer) Objects() []fyne.CanvasObject { return r.objects }


2. CUSTOM THEMES
-----------------------------------
type MyTheme struct {}

func (MyTheme) Color(n fyne.ThemeColorName, v fyne.ThemeVariant) color.Color {
    if n == theme.ColorNameButton {
        return color.RGBA{200,50,50,255}
    }
    return theme.DefaultTheme().Color(n,v)
}

func (MyTheme) Icon(n fyne.ThemeIconName) fyne.Resource {
    return theme.DefaultTheme().Icon(n)
}

func (MyTheme) Font(s fyne.TextStyle) fyne.Resource {
    return theme.DefaultTheme().Font(s)
}

func (MyTheme) Size(n fyne.ThemeSizeName) float32 {
    return theme.DefaultTheme().Size(n)
}

Apply theme:
a.Settings().SetTheme(&MyTheme{})


3. NAVIGATION STRUCTURES (ROUTER PATTERNS)
-----------------------------------
Use a single container:

router := container.NewMax()
router.Objects = []fyne.CanvasObject{ pages.Home() }

Go to another page:
router.Objects = []fyne.CanvasObject{ pages.Settings() }
router.Refresh()

Advanced router with mapping:
pages := map[string]fyne.CanvasObject{
    "home": HomePage(),
    "settings": SettingsPage(),
}

Change page:
router.Objects = []fyne.CanvasObject{pages["settings"]}


4. STATE MANAGEMENT
-----------------------------------
Global shared state pattern:

type AppState struct {

var State = AppState{}

Pass state into views:
pages.BuildDashboard(&State)

6. AUTO-REFRESH LOOPS
-----------------------------------
Ticker refresh pattern:

ticker := time.NewTicker(time.Second)
go func(){
    for range ticker.C {
        fyne.Do(func(){
            canvas.Refresh(myObj)
        })
    }
}()

Stop ticker:
ticker.Stop()


7. ASYNC TASKS (GO ROUTINES IN UI APPS)
-----------------------------------
Run heavy work:

go func() {
    result := computeSomething()
    fyne.Do(func(){
        label.SetText(result)
    })
}()

Always use fyne.Do() when updating UI from goroutines.

10. DRAG & DROP
-----------------------------------
Implement Drag-and-Drop:

type DraggableLabel struct {
    widget.Label
}

func (l *DraggableLabel) Dragged(e *fyne.DragEvent) {
    l.Move(fyne.NewPos(
        l.Position().X+e.Dragged.DX,
        l.Position().Y+e.Dragged.DY,
    ))
}

func (l *DraggableLabel) DragEnd() {}

Use like:
lbl := &DraggableLabel{}
lbl.ExtendBaseWidget(lbl)


11. CUSTOM DRAWING (CANVAS PAINTING)
-----------------------------------
Custom canvas object:

type MyCanvas struct {
    fyne.CanvasObject
}

func (m *MyCanvas) MinSize() fyne.Size {
    return fyne.NewSize(200,200)
}

func (m *MyCanvas) Resize(s fyne.Size) {}
func (m *MyCanvas) Position() fyne.Position { return fyne.NewPos(0,0) }
func (m *MyCanvas) Move(p fyne.Position) {}
func (m *MyCanvas) Show() {}
func (m *MyCanvas) Hide() {}
func (m *MyCanvas) Visible() bool { return true }

func (m *MyCanvas) Refresh(){
    canvas.Refresh(m)
}

Render inside Draw():
canvas := canvas.NewRasterWithPixels(func(x,y,int,w,h int) color.Color {
    return color.RGBA{uint8(x), uint8(y), 128, 255}
})

Use in container:
container.NewMax(myCanvasObj)

15. WORKING WITH FILES
-----------------------------------
Open file:
dialog.ShowFileOpen(func(r fyne.URIReadCloser, err error){
    if r != nil {
        data, _ := io.ReadAll(r)
        ...
    }
}, w)

Save file:
dialog.ShowFileSave(func(wr fyne.URIWriteCloser, err error){
    if wr != nil {
        wr.Write([]byte("data"))
        wr.Close()
    }
}, w)

19. PERFORMANCE TIPS
-----------------------------------
• Use raster/rasterWithPixels for heavy drawing
• Avoid refreshing entire container; refresh only changed items
• Reuse widgets when possible
• Long loops must run in goroutines with fyne.Do()
